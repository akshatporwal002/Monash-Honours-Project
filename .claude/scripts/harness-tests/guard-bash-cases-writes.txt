# Extra cases: writing a file whose CONTENT mentions a secret path is not reading a secret.
# Regression for a false positive that blocked writing a commit message describing the
# secret-read rule itself. The read rule must fire on arguments, not on heredoc bodies.
EXPECT_ALLOW|cat > msg.txt <<'EOF'
EXPECT_ALLOW|echo "denies reads of .env, .env.*, *.pem, id_rsa*, *.key" > notes.md
EXPECT_ALLOW|cat >> .gitignore <<'EOF'
EXPECT_ALLOW|git commit -F msg.txt
EXPECT_ALLOW|sed -n '1,20p' src-main/backend/app/main.py

# ...but an actual read in any segment still fires.
EXPECT_DENY|echo hi && cat .env
EXPECT_DENY|cat > out.txt <<'EOF'; cat .env
EXPECT_DENY|tail -n 20 config/prod.key
